#include<bits/stdc++.h>
#define int long long
using namespace std;

bool check(int x)
{
    int k = 0;
    while(x)
    {
        k += x % 10;
        x /= 10;
    }

    return k == 10;
}

void solve()
{
    int ans = 19;
    int k = 1;
    cin >> k;
    int cnt = 1;
    for(int i = ans; ; i++)
    {
        if(check(i))
        {
            if(cnt == k)
            {
                cout << i << "\n";
                return;
            }
            cnt++;
        }
    }
}

signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    while(t--)
        solve();
    return 0;
}

#include<bits/stdc++.h>
#define int long long
using namespace std;


void solve()
{
    int a, b, x, y;
    cin >> a >> b >> x >> y;
    int xygcd = __gcd(x, y);
    x = x / xygcd;
    y = y / xygcd;
    int l = 0, r = a + 1;
    while(l + 1 < r)
    {
        int mid = (l + r) / 2;
        if((mid * y / x) <= b) l = mid;
        else r = mid;
    }

    if((l * y) != (l / x * y * x)) 
    {
    	cout << l / x * y * x / y << " " << l / x * y << "\n";
    }
	else cout << l << " " << l / x * y << "\n";
}

signed main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int t = 1;
    while(t--)
        solve();
    return 0;
}
