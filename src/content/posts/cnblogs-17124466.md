## A.刷视频的小艾学长

- 算法标签：双指针
- 前后连续，并且保证每一个之前都有一个，然后不断缩小区间长度

```cpp
#include<bits/stdc++.h>
#define av first
#define up second
using namespace std;


int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m;
    cin >> n >> m;
    vector< pair<int, int> > a(n);
    for(int i = 0; i < n; i++) cin >> a[i].av;
    for(int i = 0; i < n; i++) cin >> a[i].up;
    sort(a.begin(),a.end());
    vector<int> vis(n+1);
    int need = 0,len = n + 1,l = -1, r = -1;
    for(int i = 0, j = 0; i < n; i++)
    {
        if(!vis[a[i].up]) need++;
        vis[a[i].up]++;
        while(j < i && vis[a[j].up] > 1)
        {
            vis[a[j].up]--;
            j++;
        }
        if(need == m && i - j + 1 < len)
        {
            len = i-j+1;
            l = j, r = i;
        }
            
    }
    
    if(l == -1) cout << "-1";
    else cout << l+1 << " " << r+1;
    return 0;
}
```

## C.**lamo的美好幻想**

- 贪心，从小到大排序时，取n-k到n位，进行顺次加t，答案为加完t后新排序后的第k大的数

```cpp
#include<bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n, m, k, t;
    cin >> n >> m >> k >> t;
    vector<int> a(n);
    for(int i = 0 ;i < n; i++) cin >> a[i];
    sort(a.begin(),a.end());
    for(int i = n - k, j = 1; i < n && j <= m ; i++, j++) a[i] += t;
    sort(a.begin(),a.end());
    int ans = a[n-k];
    cout << ans;
    return 0;
}
```

## D.**bocchi酱想交朋友!**

- 图的构建基础,将图连通

```cpp
#include<bits/stdc++.h>
using namespace std;

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    vector<int> a(n), b(n);
    for(int i = 0; i < n; i++) cin >> a[i];
    for(int i = 0; i < n; i++) cin >> b[i];
    vector<int> vis(n+1);
    vector<vector<int> > e(n+1);
    for(int i = 0; i < n; i++)
    {
        e[a[i]].push_back(b[i]);
        e[b[i]].push_back(a[i]);
    }
    
    
    function <void(int )> dfs = [&](int u)
    {
        vis[u] = 1;
        for(auto v:e[u])
        {
            if(!vis[v]) dfs(v);
        }
    };
    
    int ans = -1;
    
    for (int i = 1; i <= n; i++)
    {
        if(!vis[i]) dfs(i),ans++;
    }
    cout << ans;
    return 0;
}
```

## F.N个数求和

- 最大公约数gcd

```cpp
#include<bits/stdc++.h>
using namespace std;

typedef long long ll;
ll gcd(ll a, ll b)
{
    return b?gcd(b,a%b):a;
}

int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    
    int n;
    cin >> n;
    
    ll A = 0, B = 1;
    for(int i = 1; i <= n; i++)
    {
        ll a, b;
        cin >> a;
        // cin.ignore();
        char c;cin >> c;
        cin >> b;
        ll d = gcd(a, b);
        a /= d,b /= d;
        A = A * b + a * B,B *= b;
        d = gcd(A, B);
        A /= d;
        B /= d;
    }
    
    
    ll Z = A / B;
    A %= B;
    if(!Z && !A) puts("0");
    if(Z) cout << Z;
    if(A && Z) cout << " ";
    if(A) cout << A << '/' << B;
    return 0;
}
```

## H.连续因子

- 双指针算法
- 这里连续因子是指一个数拆成几个数相乘，里面最长的连续因子，比如630，$3*5*6*7$,最长为$5*6*7$,长度为3

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
int main()
{
    
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    ll n;
    cin >> n;
    
    ll prod = 1;
    int len = 1,l = n,r = n;
    for(int i = 2, j = 2; 1ll * (i-1) * (i-1) <= n; i++)
    {
        prod *= i;
        while(j < i && n%prod)
        {
            prod /= j;
            j++;
        }
        
        if( n % prod == 0 && (len < i - j + 1 || (len == i - j + 1 && j < l)))
        {
            len = i - j + 1;
            l = j, r = i;
        }  
    }
    
    cout << len << "\n";
    for(ll i = l; i <= r; i++)
    {
        cout << i <<"*\n"[i==r];
    }

    return 0;
}
```

## J.天梯赛座位分配
- 模拟就好，注意如果最后剩余一个队时，一定要隔一个坐安排座位

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 105;

int Team[N];
int n;
int sum;
vector<vector<int> > ans(N);
int cnt,clear;
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    cin >> n;
    for(int i = 0; i < n; i++)
    {
        cin >> Team[i];
        Team[i] *= 10;
        sum += Team[i];
    }
    while(sum)
    {
        for(int i = 0; i < n; i++)
        {
            if(Team[i])
            {
                Team[i]--;
                sum--;
                ans[i].push_back(++cnt);
                if(clear == n-1) cnt++;
                if(!Team[i]) clear++;
                
            }
        }
    }
    
    for(int i = 0; i < n; i++)
    {
        cout << '#' << i + 1 << "\n";
        for(int j = 1; j <= ans[i].size(); j++)
        {
            cout << ans[i][j - 1] << " \n" [j % 10 == 0];
        }
        
    }
    
    return 0;
}
```

## K.阅览室
- 模拟就好

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 5;
int bookNum, startTime[N],tim;
int cnt;
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n;
    cin >> n;
    memset(startTime, -1, sizeof startTime);
    while(n)
    {
        int hh,mm;
        char op,ig;
        cin >> bookNum >> op >> hh >> ig >> mm;
        if(!bookNum)
        {
            n--;
            cout << cnt << " " << (cnt == 0 ? cnt :round(1. * tim / cnt)) << "\n";
            cnt = 0, tim = 0; 
            for(int i = 1; i <= N; i++) startTime[i] = -1;
        }
        else if(op == 'S')
        {
            startTime[bookNum] = hh * 60 + mm;
        }
        else 
        {
            if(startTime[bookNum] != -1)
            {
                tim += hh * 60 + mm - startTime[bookNum];
                cnt++;
                startTime[bookNum] = -1;
            }
        }
        
        
    }
    
    return 0;
}
```

## L.整除光棍
- 除法高精度，很巧，已知有限个1构成的数能除开n,从高位到低位模拟除法，一位一位输出，注意前导零。
```cpp
#include<bits/stdc++.h>
using namespace std;


int main()
{
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    
    int n;
    cin >> n;
    int now = 0;
    int flag = 0;
    int len = 0;
    
    do{
        now = now * 10 + 1;
        len++;
        int remainder = now / n;
        if(remainder) flag = 1;
        if((flag && !remainder) || remainder)
        {
            cout << remainder;
        }
        
        now %= n;
        
    }while(now);

}
```
