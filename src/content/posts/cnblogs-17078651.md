### 6 sort与cmp

```cpp
struct sno 
{
	int x,y,z,w,d;
}q[305];


bool cmp(sno a,sno b)
{
    if(a.w>b.w) return 1;
    else if(a.w<b.w) return 0;
    else
    {
        if(a.x>b.x) return 1;
        else if(a.x<b.x) return 0;
        else
        {
            if(a.d>b.d) return 0;
            else return 1;
        }
    }
}


//主函数：sort(q,q+n,cmp);
```
