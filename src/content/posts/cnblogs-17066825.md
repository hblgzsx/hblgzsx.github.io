# 二分入门注意点

- 理解：用分界线去理解

- 一开始分界线在0和1之间、n和n+1之间

- 所以l=0,r=n+1

- 二分必须满足单调性

- ```cpp
  模板1 ---分界线写法：
  int l=0,r=n+1;
  while(l+1<r){
      int mid=(l+r)>>1;
      if(a[mid]<=x) l=mid;
      else r=mid;
  }
  
  
  模板2 ---整数二分普通写法：
  while条件中为l<r,区间向左收缩
  int l=1,r=n;
  while(l<r){
      int mid = (l+r)>>1;
      if(a[mid]<=x) r=mid;
      else l=mid+1;
  }
  
  while条件中为l<r,区间向左收缩
  int l=1,r=n;
  while(l<r){
      int mid = (l+r+1)>>1;
      if(a[mid]<=x) l=mid;
      else r=mid-1;
  }
  
  while条件中为l<=r,补充写法
  int l=1,r=n;
  while(l<=r){
      int mid = (l+r)>>1;
      if(a[mid]<=x) r=mid-1;
      else l=mid+1;
  }
  
  tips:浮点数二分时，条件都为l=mid,r=mid
  ```

## 1.二分查找

[题目来源](http://oj.daimayuan.top/problem/59)

### 题目描述

![image-20230122182535282](/images/cnblogs/353a21c237de97351923.png)

### 提交代码

```cpp
#include<bits/stdc++.h>
using namespace std;

const int N=2e5+5;

int n,m,a[N];

int solve(int x){
	int l=0,r=n+1;
	while(l+1<r){
		int mid=(l+r)>>1;
		if(a[mid]<x) l=mid;
		else r=mid;
	}
	return l;
}

int solve1(int x){
	int l=0,r=n+1;
	while(l+1<r){
		int mid=(l+r)>>1;
		if(a[mid]<=x) l=mid;
		else r=mid;
	}
	return l;
}
int main(){
	scanf("%d%d",&n,&m);
	for(int i=1;i<=n;i++) scanf("%d",&a[i]);
	sort(a+1,a+n+1);
	while(m--){
		int x;scanf("%d",&x);
		int A=solve(x),B=solve1(x);
		printf("%d %d %d\n",A,B-A,n-B);
	}
	return 0;
}
```

## 2.二分答案

### 题目描述

![image-20230122183423668](/images/cnblogs/05203c265fbce58253cb.png)

### 提交代码

- 提示用法，0ll代表long long 数据类型的零，1ll*int数据，代表转换为long long 数据类型

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const int N=1e5+10;
int n,a[N];
ll k;
ll solve(ll x){
	ll res=0;
	for(int i=1;i<=n;i++) res+=max(0ll,x-a[i]);
	return res;
}
int main(){
	scanf("%d%lld",&n,&k);
	for(int i=1;i<=n;i++) scanf("%d",&a[i]);
	ll l=1,r=1e14;
	while(l+1<r){
		ll mid =(l+r)>>1;
		if(solve(mid)<=k) l=mid;
		else r=mid;
	}
	printf("%lld\n",l);
	return 0; 
}
```

## 3.合并序列2

[题目来源](http://oj.daimayuan.top/course/16/problem/649)

### 题目描述

![image-20230122190906712](/images/cnblogs/700f4bc3621bcf2da4e5.png)

### 提交代码

```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

const int N=1e5+10;
int n,k[N],b[N],m;

ll calc(int x){
	ll res = 0;
	for(int i=1;i<=n;i++){
		if(x>=b[i]) res+=(x-b[i])/k[i]+1;
	}
	return res; 
}

ll sum(int x){
	ll res=0;
	for(int i=1;i<=n;i++){
		if(x>=b[i]){
			int y = (x-b[i])/k[i]+1;
			res+=1ll*(b[i]+(y-1)*k[i]+b[i]) * y/2;//等差数列前n项和，(a1+an)*项数/2
		}
	}
	return res;
}


int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;i++){
		scanf("%d%d",&k[i],&b[i]);
	}
	scanf("%d",&m);
	int l=0,r=1e9;
	while(l+1<r){
		int mid=(l+r)>>1;
		if(calc(mid)<=m) l=mid;
		else r=mid;

	}
	printf("%lld\n",sum(l)+1ll*(m-calc(l))*(l+1));//不足m个时，补上缺少的
}
```

## 4.快速集合

[题目来源](http://oj.daimayuan.top/course/16/problem/648)

### 题目描述

![image-20230125110022983](/images/cnblogs/a7123bea837f632f53c6.png)

### 提交代码

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+10;
int n,a[N],b[N];

bool check(double x){
	double L=-1e10,R=1e10;
	for(int i=1;i<=n;i++){
		double l = a[i]-b[i]*x,r=a[i]+b[i]*x;
		L=max(L,l);
		R=min(R,r);
	}
	if(L>R) return 0;
	return 1;
}


int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;i++) scanf("%d%d",&a[i],&b[i]);
	double l=0,r=5e5;
	for(int i=1;i<=100;i++){
		double mid = (l+r)/2;
		if(check(mid)) r=mid;
		else l=mid;
	}
	printf("%.10f",r);
	return 0;
}

```

## 5.最长上升子序列

### 题目描述

![image-20230125114645234](/images/cnblogs/6c2d42f015806c39559f.png)

### 提交代码

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=1e5+10;
int n,a[N],c[N],y;
//c[]数组永远放的是连续的只对当前序列有用的子序列
int main(){
	scanf("%d",&n);
	for(int i=1;i<=n;i++ ) scanf("%d",&a[i]);
	y=0;
	for(int i=1;i<=n;i++){
		if(a[i]>c[y]) c[++y] = a[i];
		else{
			int l=0,r=y;
			while(l+1<r){
				int mid=(l+r)>>1;
				if(c[mid]<a[i]) l=mid;
				else r=mid;
			}
			c[r]=a[i];
		}
	}

	printf("%d\n",y);
}
```
