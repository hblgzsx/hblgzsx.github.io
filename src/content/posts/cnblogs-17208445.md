```cpp
// Problem: B - Trimmed Mean
// Contest: AtCoder - AtCoder Beginner Contest 291（Sponsored by TOYOTA SYSTEMS）
// URL: https://atcoder.jp/contests/abc291/tasks/abc291_b
// Memory Limit: 1024 MB
// Time Limit: 2000 ms
// 
// Powered by CP Editor (https://cpeditor.org)

#include<bits/stdc++.h>
using namespace std;

int main()
{
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	
	int n;
	cin >> n;
	int k = 0;
	vector<int> a(5 * n + 10);
	while(cin >> a[k++]);
	k--;
	sort(a.begin(), a.begin() + k);
	long long sum = 0;
	for(int i = n; i < (k - n); i++)
	{
		sum += a[i];
		// cout << sum << "\n";
	}
	double ans = sum * 1./ (k - 2 * n);
	cout << fixed << setprecision(7) << ans << "\n";
	
	return 0;
}
```

```cpp
// Problem: C - LRUD Instructions 2

#include<bits/stdc++.h>
using namespace std;

map<int, map<int, int> > ans;
int x, y;
bool check(char s)
{
	if(s == 'U') y++;
	if(s == 'D') y--;
	if(s == 'L') x--;
	if(s == 'R') x++;
	ans[x][y]++;
	if(ans[x][y] == 2) return true;
	return false;
} 
int main()
{
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	int n;
	cin >> n;
	string str;
	cin >> str;
	ans[0][0] = 1;
	for(int i = 0; i < n; i++)
	{
		if(check(str[i]))
		{
			cout << "Yes\n";
			return 0;
		}
	}
	cout << "No\n";
	return 0;
}
```

# D.[Flip Cards](https://atcoder.jp/contests/abc291/tasks/abc291_d)（dp）

```cpp
// Problem: D - Flip Cards
// Contest: AtCoder - AtCoder Beginner Contest 291（Sponsored by TOYOTA SYSTEMS）
// URL: https://atcoder.jp/contests/abc291/tasks/abc291_d
// Memory Limit: 1024 MB
// Time Limit: 2000 ms
// 
// Powered by CP Editor (https://cpeditor.org)

#include<bits/stdc++.h>
#define mod 998244353
using namespace std;

int main()
{
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	
	auto solve = [&]()
	{
		int n;
		cin >> n;
		vector< vector<int> > ab(n, vector<int>(2));
		
		for(int i = 0; i < n; i++)
		{
			cin >> ab[i][0] >> ab[i][1];
		}
		vector<int> dp(2,1);
		
		int z[] = {0,1};
		for(int i = 1; i < n; i++)
		{
			vector<int> pdp(2,0);
			for(auto x:z)
			{
				for(auto y :z)
				{
					if(ab[i - 1][x] != ab[i][y])
					{
						(pdp[y] += dp[x]) %= mod;
					}
				}
			}
			swap(dp,pdp);
		}
		
		cout << (dp[0] + dp[1]) % mod << "\n";
	};
	solve();
	return 0;
}
```

# E.[Find Permutation](https://atcoder.jp/contests/abc291/tasks/abc291_e)(拓扑排序)

```cpp
// Problem: E - Find Permutation
// Contest: AtCoder - AtCoder Beginner Contest 291（Sponsored by TOYOTA SYSTEMS）
// URL: https://atcoder.jp/contests/abc291/tasks/abc291_e
// Memory Limit: 1024 MB
// Time Limit: 2000 ms
// 
// Powered by CP Editor (https://cpeditor.org)

#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10, M = N << 1;

int n, m, d[N], ans[N], res[N];
int h[N], e[M], w[M], ne[M], idx;
bool vis[N];

void add(int a, int b)
{
	e[idx] = b, ne[idx] = h[a], h[a] = idx++;
} 
int main()
{
	ios::sync_with_stdio(false);
	cin.tie(nullptr);
	cin >> n >> m;
	memset(h, -1, sizeof h);
	for(int i = 1; i <= m; i++)
	{
		int u, v;
		cin >> u >> v;
		add(u, v);
		d[v]++;
	}
	
	queue<int> q;
	bool flag = true;
	int cnt = 0;
	for(int i = 1; i <= n; i++)
	{
		if(!d[i]) q.push(i);
	}
	
	while(q.size())
	{
		if(q.size() > 1)
		{
			flag = false;
			break;
		}
		
		auto t = q.front();
		q.pop();
		ans[++cnt] = t;
		for(int i = h[t]; i != -1; i = ne[i])
		{
			int j = e[i];
			d[j]--;
			if(!d[j]) q.push(j);
		}
	}
	if(cnt != n)  cout << "No\n";
	else
	{
		cout << "Yes\n";
		for(int i = 1; i <= n; i++)
		{
			res[ans[i]] = i;
		}
		for(int i = 1; i <= n; i++)
		{
			cout << res[i] << " \n"[i == n];
		}
	}
	return 0;
}
```
